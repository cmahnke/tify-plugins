Rendered MusicXML notation, aligned onto the scan, with playback.

Code example 
<!-- Viewer and panel hosts in your markup -->
<div id="tify"></div>
<aside id="panel"></aside>
import Tify from 'tify';
import { musicxmlPlugin } from '@projektemacher/tify-plugin-musicxml';

const tify = new Tify({ container: '#tify', manifestUrl });
const plugin = await musicxmlPlugin(tify, {
  panelEl: document.querySelector('#panel'),
  musicxml: 'https://example.org/page.musicxml',
});
