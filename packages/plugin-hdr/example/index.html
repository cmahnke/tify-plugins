<!-- Only the viewer needs markup — the plugin creates its own panel -->
<div id="tify"></div>
import Tify from 'tify';
import { enableHdrCanvasOnce, hdrPlugin } from '@projektemacher/tify-plugin-hdr';

// CRITICAL: hook the canvas before Tify creates its OpenSeadragon canvas.
const hdrHook = enableHdrCanvasOnce();
const tify = new Tify({ container: '#tify', manifestUrl });
// No panelEl: the plugin creates its own panel and docks
// a toggle button into the viewer header.
const plugin = await hdrPlugin(tify, {
  hook: hdrHook,
});
