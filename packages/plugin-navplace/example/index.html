Places

Places from the IIIF navPlace extension on a configurable vector-tile map.

Code example 
<!-- Viewer and panel hosts in your markup -->
<div id="tify"></div>
<aside id="panel"></aside>
import Tify from 'tify';
import { navPlacePlugin } from '@projektemacher/tify-plugin-navplace';

const tify = new Tify({ container: '#tify', manifestUrl });
const plugin = await navPlacePlugin(tify, {
  panelEl: document.querySelector('#panel'),
  // Basemap (OpenFreeMap by default):
  style: 'https://tiles.openfreemap.org/styles/positron',
});
